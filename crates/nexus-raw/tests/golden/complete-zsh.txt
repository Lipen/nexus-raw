_nxr_commands() {
    local commands; commands=(
'get:GET a URL to a file or stdout' \
'put:PUT a file, optionally with its sha-sibling marker' \
'head:HEAD a URL\: status, size, content type' \
'sha:The sha256 of a file or URL' \
'up:Upload a local directory (§5.2)' \
'down:Download a remote directory into a local one (§5.2, §5.2.1)' \
'diff:Compare a local directory against a remote one\: the delta report' \
'rm:Delete the enumerated names from a remote directory (§5.4)' \
'point:Delete a pointer file (§5.4)' \
'mirror:Pour a version from one repository into another\: bytes, markers, the version document' \
'mv:Move a version\: mirror it into the destination, then delete the same names at the source' \
'ls:List the entries of a raw directory URL, at any tree depth (--assets\: flat artifact names)' \
'channel:Read or write a channel ref\: a token file with any name' \
'verify:Check local bytes, markers and digests (no network)' \
'doctor:Diagnose credentials, TLS and reachability' \
'service:Server metadata through the Sonatype service REST API (not part of the storage protocol)' \
'complete:Print a shell completion script to stdout (bash, zsh, fish, powershell)' \
'help:Print this message or the help of the given subcommand(s)' \
    )
    _describe -t commands 'nxr commands' commands "$@"
}
